I use these two definitions of nat_fold_right and nat_parafold_right, adapted from Professor Olivier Danvy's lecture notes, throughout the rest of my code.

(* nat_fold_right *)
let nat_fold_right zero_case succ_case n_given =
  (* nat_fold_right : 'a -> ('a -> 'a) -> int -> 'a *)
  let () = assert (n_given >= 0) in
  let rec visit n =
    if n = 0
    then zero_case
    else let n' = n - 1
      in let ih = visit n'
      in succ_case ih    (* <-- succ_case takes one argument *)
  in visit n_given;;

(* nat_parafold_right *)
let nat_parafold_right zero_case succ_case n_given =
  (* nat_parafold_right : 'a -> (int -> 'a -> 'a) -> int -> 'a *)
  let () = assert (n_given >= 0) in
  let rec visit n =
    if n = 0
    then zero_case
    else let n' = n - 1
      in let ih = visit n'
      in succ_case n' ih    (* <-- succ_case takes two arguments *)
  in visit n_given;;

I also use the following two definitons to conduct positive random tests, i.e. testing whether the function in question is the same as a witness function (that we know is correct) for a bunch of random inputs.

let show_bool b =

(* positive random test *)
let positive_random_test_for_unary_function_taking_an_integer_once r name candidate witness show_result =
  let n = Random.int r
  in let actual_result = candidate n
  and expected_result = witness n
  in if actual_result = expected_result
  then true
  else let () = Printf.printf (* error message *)

    in false;;

let repeat n_given test candidate =
  let () = assert (n_given >= 0) in

    if n = 0
    then true
    else test candidate && visit (n - 1)
  in visit n_given;;

Q. Define the factorial function using nat_fold_right.

Exercise 3 asks us to implement the factorial function. First, it's a good idea to have a unit test for whatever factorial function we implement!

(* a unit test for our future factorial function *)
let test_fac candidate =
  (* the base case: *)
  let b0 = (candidate 0 = 1)
  (* some intuitive cases: *)
  and b1 = (candidate 1 = 1)
  and b2 = (candidate 2 = 2)
  and b3 = (candidate 3 = 6)
  and b4 = (candidate 4 = 24)
  and b5 = (candidate 5 = 120)
  (* instance of the induction step: *)
  and b6 = (let n = Random.int 20
            in candidate (succ n) = (succ n) * candidate n)
  (* etc. *)
  in b0 && b1 && b2 && b3 && b4 && b5 && b6;;

Perfect! Now we know whether our future factorial function will horribly fail or not (mostly)! And, now to actually get on with our implementation.

Using nat_parafold_right may be a good start! This is because to calculate the factorial of n, we multiply n by the factorial of (n - 1). Using nat_parafold_right will give us access to n at each stage of our computation.

(* factorial function using nat_parafold_right *)
let fac_npfr n =
  let () = assert (n >= 0) in
  nat_parafold_right 1 (fun n' ih -> (succ n') * ih) n;;

(* testing function with unit test *)
let () = assert (test_fac fac_npfr);;

And as we can see, it works!

Let's think about how it works, by trying to calculate the factorial of 5 the same way the function would:

  1. To calculate the factorial of 5, we must multiply 5 by the factorial of 4. For that, we must calculate the factorial of 4.
  2. To calculate the factorial of 4, we must multiply 4 by the factorial of 3. For that, we must calculate the factorial of 3.
  3. To calculate the factorial of 3, we must multiply 3 by the factorial of 2. For that, we must calculate the factorial of 2.
  4. To calculate the factorial of 2, we must multiply 2 by the factorial of 1. For that, we must calculate the factorial of 1.
  5. To calculate the factorial of 1, we must multiply 1 by the factorial of 0. We know, because the base case is the factorial of 0, that the factorial of 0 is 1. Hence, we can pass 1 * 1 back to the above calculation.
  6. To calculate the factorial of 2, we must multiply 2 by the factorial of 1. We multiply 2 * 1 and pass 2 to the calculation before us.
  7. To calculate the factorial of 3, we must multiply 3 by the factorial of 2. We multiply 3 * 2 and pass 6 to the calculation before us.
  8. To calculate the factorial of 4, we must multiply 4 by the factorial of 3. We multiply 4 * 6 and pass 24 to the calculation before us.
  9. To calculate the factorial of 5, we must multiply 5 by the factorial of 4. We multiply 5 * 24 and return 120.

And, the same thing is happening here!
For every run of succ_case, our function multiplies the current value of n by the factorial of n' (which in this case is ih). This happens until we reach the factorial of 0, a factorial whose value we know is 1!
Over here, we pass "1" to nat_parafold_right as it is the factorial of 0, and, hence, it represents our zero_case (that is, our base case).

Now, we can try implementing the factorial function using nat_fold_right, like below:

(* factorial function using nat_fold_right *)
let fac_nfr n =
  let () = assert (n >= 0) in
  let (n, result) =
    nat_fold_right (0, 1) (fun (n', ih) -> ((succ n'), (succ n') * ih)) n
  in result;;

(* testing function with unit test *)
let () = assert (test_fac fac_nfr);;

Over here, (n, result) represent the given n and the result of our calculation, and we only return the result.

We begin with (0, 1), that is our base case, because:

  • 0: we must iterate from 0 until the given n
  • 1: the factorial of 0 (0!), the base case, is 1

The input function simply iterates ahead through "(succ n')", and we use "(succ n') * ih" to calculate the actual factorial, (similar to how we did in our implementation with parafold) hence, by using "(succ n')" to iterate, we get access to n' as well, which we must use to calculate our factorial!

2.4.1 Q. Compose a unit-test function for sum, based on its inductive specification.

Hmm. Unit tests. We're used to those! BUT, how do we make the unit test inductive?
We test the whether the function works for the inductive step or not, of course!

Let's look at the test below:

(* a. a unit test for our future sum function *)
let test_sum candidate f =
  (* base case *)
  let b0 = (candidate f 0 = f 0)
  (* some intuitive cases *)
  and b1 = (candidate f 1 = f 0 + f 1)
  and b2 = (candidate f 2 = f 0 + f 1 + f 2)
  (* some instances of successive steps *)
  and b3 = (candidate f 3 = candidate f 2 + f 3)
  and b4 = (candidate f 4 = candidate f 3 + f 4)
  (* instance of the induction step: *)
  and b5 = (let n = Random.int 20
            in candidate f n + f (succ n) = candidate f (succ n))
  in b0 && b1 && b2 && b3 && b4 && b5;;

As we can see, apart from our intuitive cases and the base case, we also have some instances where the the inductive step is tested when going from 2 to 3 and 3 to 4. Also, we test the inductive step on some random random numbers!
With the inductive step, we test whether a) applying the candidate function to n, and b) applying the candidate function to n' and adding f n to it, are the same (a = b)!
(Do note, in this case, instead of n', I've used succ n. The above expression could also be written as candidate f n' + f n = candidate f n, where n' represents n - 1.

2.4.2 Q. Implement this specification as a structurally recursive function expecting a function from int to int and a non-negative integer.

Let's think about what an inductive specification would look like for the sumtorial function. Perhaps, we can take inspiration from the inductive step in the unit test.

  • base case: n = 0, we return the value yielded by applying the given function to 0.
  • inductive step (getting n in terms of n'): the answer to sum f n in terms of n' is f (n' + 1) + sum f n'.

    if n = 0
    then f 0
    else let n' = n - 1
      in let ih = visit n'
      in f (succ n') + ih
  in visit n_given;;

2.4.4 Q. Express your implementation using either nat_fold_right or nat_parafold_right, your choice. Justify this choice, and verify that your new implementation passes your unit test.

As we discussed earlier, our base case and our inductive step are:

  • base case: n = 0, we return the value yielded by applying the given function to 0.
  • inductive step (getting n in terms of n'): the answer to sum f n in terms of n' is f (n' + 1) + sum f n'.

Hence, we see that our inductive step, which is implemented by our succ_case function, requires us to know what n or n' is.
Hence, it makes more sense for us to use nat_parafold_right rather than nat_fold_right.

Hence, we can subsitute our zero_case (that is, our base case) and our succ_case function in nat_parafold_right as below:

(* d. recursive function for sum using nat_parafold_right *)
let sum_npfr f n =
  nat_parafold_right (f 0) (fun n' ih -> f (succ n') + ih) n;;

(* testing function with unit test *)
let () = assert (test_sum sum_npfr (fun x -> 2 * x));;

And, of course, the function passes our unit test.

That being said, we can still implement the function with nat_fold_right as well (later we'll see how nat_fold_right and nat_parafold_right are really the same, stay tuned!):

(* similar to exercise 03, using nat_fold_right *)
let sum_nfr f n =
  let (n, result) =
    nat_fold_right (0, (f 0)) (fun (n', ih) -> (succ n', (f (succ n') + ih))) n
  in result;;

(* testing function with unit test *)
let () = assert (test_sum sum_nfr (fun x -> 2 * x));;

Over here, (n, result) represent the given n and the result of our calculation, and we only return the result.

We begin with (0, (f 0)), that is our base case, because:

  • 0: we must iterate from 0 until the given n
  • 0: for our base case, we return the result of applying the given function to 0

The input function simply iterates ahead through "(succ n')", and we use "f (succ n') + ih" to calculate the subsequent result (similar to how we did in our implementation with parafold). Hence, by using "(succ n')" to iterate, we get access to n' as well, which we must use to calculate our result!

And, of course, the function passes our unit test.

We beGin with (0, 1), that is our base case, because:

  • 0: we must iterate from 0 until the given n
  • 1: the sum of the first odd number, 1, is 1

The input function simply iterates ahead through (succ n'), and we use (make_odd_number (succ n')) + ih to calculate the actual result, (similar to how we did in our implementation with parafold) hence, by using "(succ n')" to iterate, we get access to n' as well, which we must use to calculate our result!

2.7.3 Implement the three predicates as mutually recursive function

Let's think about how we would implement the ternary predicates as mutually recurisve functions!

Well, let's first imagine a scenario where we call the ternary predicate on a number, n:

  • if the n is 0, we can return true as we know that 0 is a ternary number!
  • however, if the number is not 0, we can simpyl test whether n-1 is a pre-ternary number! if n-1 is a pre-ternary number, then n must be a ternary number! so, we can simply call the pre-ternary predicate on n-1!
  • now, for calling the pre-ternary predicate on n-1: if n-1 is 0, we can return false, as 0 is not a pre-ternary number, and, therefore, n cannot be a ternary number, as 1 is not a ternary number! however, if n-1 is not 0, we can simply test whether n-2 is a post-ternary number! if n-2 is a post-ternary number, n-1 must be a pre-ternary number, and n must be a ternary number! hence, we can simply call the post-ternary predicate on n-2
  • now, for calling the post-ternary predicate on n-2: if n-2 is 0, we can return false, as 0 is not a post-ternary number, and, therefore, n cannot be a ternary number, as 2 is not a ternary number! however, if n-2 is not 0, we can simply test whether n-3 is a ternary number! if n-3 is a ternary number, n-2 must be a post-ternary number, n-1 must be a pre-ternary number, and n must be a ternary number! hence, we can simply call the ternary predicate on n-3

3.4 question 04

Q. Are the following functions equivalent?:

(fun x1 -> fun x2 -> e0) e1 e2
(fun (x1, x2) -> e0) (e1, e2)

The curried function (the haskell kind, not the indian culinary delight):

to evaluate "(fun x1 -> fun x2 -> e0) e1 e2":
application associates to the left, hence the above is equivalent to ((fun x1 -> fun x2 -> e0) e1) e2
to evaluate "((fun x1 -> fun x2 -> e0) e1) e2", we must first evaluate e2 (from what we learnt in question 02)
  - if evaluating e2 does not terminate then the entire evaluation does not terminate
  - if evaluating e2 raises an error, then the entire evaluation raises this error
  - if evaluating e2 evaluates to v2 with type t2, then
    - we evaluate ((fun x1 -> fun x2 -> e0) e1)
    - to evaluate ((fun x1 -> fun x2 -> e0) e1), we must first evaluate e1:
      - if evaluating e1 does not terminate then the entire evaluation does not terminate
      - if evaluating e1 raises an error, then the entire evaluation raises this error
      - if evaluating e1 evaluates to v1 with type t1, then
        - evaluating ((fun x1 -> fun x2 -> e0) e1) reduces to evaluating (fun x2 -> e0) in an environment where x1 denotes v1
          - evaluating (fun x2 -> e0) returns a function, then
          - evaluating ((fun x1 -> fun x2 -> e0) e1) e2 reduces to applying "this" function (the function we got from the previous line) to v2
            - applying "this" function reduces to evaluating e0 in an environment where x2 denotes v2.
            - because "this" function was evaluated in an environment where x1 denoted v1, the environment of e0 also contains x1 denoting v1.

The uncurried function:

to evaluate "(fun (x1, x2) -> e0) (e1, e2)", we must first evaluate (e1, e2)
  - to evaluate (e1, e2), we first evaluate e2 (from what we learnt in question 03)
    - if evaluating e2 does not terminate then the entire evaluation does not terminate
    - if evaluating e2 raises an error, then the entire evaluation raises this error
    - if evaluating e2 evaluates to v2 with type t2, then we evaluate e1
      - if evaluating e1 does not terminate then the entire evaluation does not terminate
      - if evaluating e1 raises an error, then the entire evaluation raises this error
      - if evaluating e1 evaluates to v1 with type t1, then
        - the result of (e1, e2) is the pair (v1, v2), then
          - we evaluate (fun (x1, x2) -> e0)
            - evaluating (fun (x1, x2) -> e0) returns a function, then
            - evaluating "(fun (x1, x2) -> e0) (e1, e2)" reduces to applying "this" function (the function we got from the previous line) to (v1, v2)
              - applying the function "this" to (v1, v2) reduces to evaluating e0 in an environment where x1 denotes v1 and x2 denotes v2

To sum up, in both cases:

  • e2 is the component that is evaluated first, and, hence,
    • if e2 diverges, then in both cases the evaluation does not terminate
    • if e2 raises an error, in both cases the evaluation raises that errors
    • if e2 has a side effect, in both cases e2 makes the same side effect
    • if e2 yields a value v2, the value is the same in both cases
  • e1 is the component that is evaluated next, and, hence,
    • if e1 diverges, then in both cases the evaluation does not terminate
    • if e1 raises an error, in both cases the evaluation raises that errors
    • if e1 has a side effect, in both cases e1 makes the same side effect
    • if e1 yields a value v1, the value is the same in both cases
  • finally, if e1 and e2 yielded v1 and v2 respectively, Then
    • e0 is evaluated in an environment where x1 denotes v1 and x2 denotes v2

Hence, each of these expressions give rise to the same evaluation/computation.
Therefore, they are equivalent!

And, so we recall the story of nat_fold_right and nat_parafold_right.

3.9.1 For any pure expression v of type int, would it be valid to simplify v * 0 into 0?

The case of v * 0:

- first of all, we note that the expressions "v" and "0" are pure.
- evaluating "v * 0" requires first evaluating 0. evaluating 0 yields 0. then evaluating "v * 0" requires evaluating v.
- since v is a pure expression, evaluating v, a variable of type int, silently yields its value in the current environment, v1, an Integer.
- then, evaluating "v * 0" reduces to multiplying 0 by v1.
- since 0 is absorbing for multiplcation, 0 multiplied by v1 evaluates to 0.
hence, the result of evaluating "v * 0" is 0.

The case of 0:

- evaluating "0" yields 0.
- hence, the result of evaluating "0" is 0.

To sum up, in both cases:

  • the evaluation silently yields 0.

Hence, it is valid to simplify 0 * v into 0 as they both are observationally equivalent, given that v is pure, of course.

3.9.2 For any potentially impure expression e of type int, would it be valid to simplify e * 0 into 0?

The case of e * 0:

(an_int 14) * 0;;

processing 14...
- : int = 0

The case of 0:

0;;

- : int = 0

Hence, it is invalid to simplify e * 0 to 0, if e is potentially impure; as there exists an e, namely (anint 14) such that e * 0 and 0 are not observationally equivalent.

3.9.3 For any pure expression v of type int, would it be valid to simplify v * 1 into v?

The case of v * 1:

- first of all, we note that the expressions "v" and "1" are pure.
- evaluating "v * 1" requires first evaluating 1. evaluating 1 yields 1. then evaluating "v * 1" requires evaluating v.
- since v is a pure expression, evaluating v, a variable of type int, silently yields its value in the current environment, v1, an Integer.
- then, evaluating "v * 1" reduces to multiplying 1 by v1.
- the identity property of multiplication states that any number multiplied by 1 keeps its identity, i.e. the number stays the same. hence, 1 multiplied by v1 evaluates to v1.

The case of v:

5.2 first degree

From Professor Olivier Danvy's lecture notes:

Given two coefficients a1 and a0, the following function returns a polynomial function of degree 1:

let make_polynomial_1 a1 a0 x =
  a1 * x + a0;;

Given such a polynomial function of degree 1, we want to compute its coefficients. So your task, should you choose to accept it, but hey, it’s not like this mission is impossible, witness the solution just below, so there really is no reason to digress here, is to implement a function crack1 of type:

(int -> int) -> int * int

that passes the following unit test:

let test_crack_1_once candidate =
  let a1 = random_int ()
  and a0 = random_int ()
  in let p1 = make_polynomial_1 a1 a0
     in let expected_result = (a1, a0)
        and ((a1', a0') as actual_result) = candidate p1
        in if actual_result = expected_result
           then true
           else let () = Printf.printf
                           "test_crack_1_once failed for %i * x^1 + %i with (%i, %i)\n"
                           a1 a0 a1' a0'
                in false;;

In words: to test a candidate cracking function,

  • we manufacture a first-degree polynomial function using two random integers;
  • we apply the candidate cracking function to this polynomial function;
  • if the resulting two integers are the same as the two random integers, the candidate cracking function passes this test;
  • otherwise, the candidate cracking function does not pass this test and an error message is emitted.

This being said, by Professor Olivier Danvy, we can move on to our own functions!

For each subsequent degree, our function will accept a function that itself accepts one more integer, and return a tuple with just one more integer as well. The additional integer in the tuple will represent the additional coefficient.

And, so, ready, set, go!

For our first cracking function we can just calculate a1 (i.e. the coefficient of x1) by subtracting p1 0!

And, so we have:

  and a0 = p1 0
  in (a1, a0);;

let () = assert (test_crack_1 crack_1);;

let crack_1_v1 p1 =

5.8 higher degree to lower degree

One final thing; we can use any polynomial cracking function to crack a polynomial of a lower degree!

That is to say, a "degree 5 cracker" can also crack polynomials of degree 4, 3, 2, 1, and 0. Similarly, a "degree 4 cracker" can also crack polynomials of degree 3, 2, 1, and 0.

Hence, there is no reason to ask individually whether a "degree 3 cracker" can crack a polynomial of degree 2 or 1, as we can see from the above rule that a higher degree cracker can always crack a lower degree polynomial.

We do something similar when we implement nat_parafold_right in terms of nat_fold_right: we ignore one of the values.

Like so:

(* using crack_5 to implement crack_0 *)
let crack_0_using_crack_5 p0 =
  let (a5, a4, a3, a2, a1, a0) = crack_5 p0
    in a0;;

let () = assert (test_crack_0 crack_0_using_crack_5);;

The value of a5, a4, a3, and a2 will simply be 0. Hence, we can simply throw those values out and only return the relevant ones.

Similarly, for crack_1, crack_2, crack_3, crack_4:

(* using crack_5 to implement crack_1 *)
let crack_1_using_crack_5 p1 =
  let (a5, a4, a3, a2, a1, a0) = crack_5 p1
    in (a1, a0);;

let () = assert (test_crack_1 crack_1_using_crack_5);;

(* using crack_5 to implement crack_2 *)
let crack_2_using_crack_5 p2 =
  let (a5, a4, a3, a2, a1, a0) = crack_5 p2
    in (a2, a1, a0);;

let () = assert (test_crack_2 crack_2_using_crack_5);;

(* using crack_5 to implement crack_3 *)
let crack_3_using_crack_5 p3 =
  let (a5, a4, a3, a2, a1, a0) = crack_5 p3
    in (a3, a2, a1, a0);;

let () = assert (test_crack_3 crack_3_using_crack_5);;

(* using crack_5 to implement crack_4 *)
let crack_4_using_crack_5 p4 =
  let (a5, a4, a3, a2, a1, a0) = crack_5 p4
    in (a4, a3, a2, a1, a0);;

let () = assert (test_crack_4 crack_4_using_crack_5);;

So, there we have a hard and fast rule to answer the many subsidary questions, all at once; once again, many birds with one stone!
